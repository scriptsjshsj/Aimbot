SliderHandle.Parent = SliderLine

local isDragging = false
local dragStartPos = Vector2.new(0, 0)
local handleStartPos = Vector2.new(0, 0)

SliderHandle.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        isDragging = true
        dragStartPos = Vector2.new(input.Position.X, input.Position.Y)
        handleStartPos = Vector2.new(SliderHandle.Position.X.Offset, SliderHandle.Position.Y.Offset)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if isDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local deltaX = input.Position.X - dragStartPos.X
        local newWidth = math.clamp(handleStartPos.X + deltaX, 0, SliderLine.AbsoluteSize.X - 10)
        SliderHandle.Position = UDim2.new(0, newWidth, 0, -5)
        
        -- Update smoothness (inverse relationship for UI feel: right is smoother)
        local percent = newWidth / (SliderLine.AbsoluteSize.X - 10)
        Config.Smoothness = 1.0 + (percent * 5) -- Range from 1 to 6
        SliderLabel.Text = string.format("Smoothness: %.1f", Config.Smoothness)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        isDragging = false
    end
end)

-- Doxing Logic (Simulated/Local)
local function doxPlayer(player)
    local info = {
        Name = player.Name,
        UserID = player.UserId,
        JoinDate = player.CreatedOn,
        RobloxAvatar = "https://www.roblox.com/headshot-thumbnail/image?userId=" .. player.UserId .. "&width=420&height=420&format=png",
        LocalIP = "127.0.0.1", -- Simulated local IP since Roblox doesn't expose real IPs easily in client scripts
        Location = "Unknown (Client-Side Script)",
        Device = "PC/Web/Mobile",
        LastSeen = os.date("%Y-%m-%d %H:%M:%S")
    }
    
    -- Print to output as if we are logging it
    print("[DOX] Target Acquired: " .. player.Name)
    print("[DOX] Data: " .. HttpService:JSONEncode(info))
    
    return info
end

-- Aimbot Logic
local function getClosestTarget()
    local closest = nil
    local minDistance = math.huge
    local myChar = game.Players.LocalPlayer.Character
    if not myChar then return nil end
    
    local myHead = myChar:FindFirstChild("Head")
    if not myHead then return nil end
    
    local myPos = myHead.Position
    
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= game.Players.LocalPlayer and player.Character and player.Character.PrimaryPart then
            local targetPos = player.Character.PrimaryPart.Position
            local distance = (myPos - targetPos).Magnitude
            
            if distance < Config.FOV and distance < minDistance then
                minDistance = distance
                closest = player
            end
        end
    end
    return closest
end

RunService.RenderStepped:Connect(function()
    if Config.AimbotEnabled then
        local target = getClosestTarget()
        
        if target and target.Character and target.Character:FindFirstChild("Head") then
            local head = target.Character.Head
            local targetCFrame = head.CFrame
            
            -- Smooth interpolation
            local tweenInfo = TweenInfo.new(Config.Smoothness, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
            local tween = TweenService:Create(game.Players.LocalPlayer.CameraSubject, tweenInfo, {
                CFrame = CFrame.new(game.Players.LocalPlayer.CameraSubject.CFrame.Position, targetCFrame.Position)
            })
            tween:Play()
            
            -- Visual indicator (optional, changes camera subject briefly to show lock)
            -- In a full exploit, this might weld a part to the head
        end
    end
end)

-- Periodic Dox simulation
while true do
    wait(10) -- Every 10 seconds, simulate finding a new person
    local randomPlayer = Players:GetPlayers()[math.random(1, #Players:GetPlayers())]
    if randomPlayer and randomPlayer ~= game.Players.LocalPlayer then
        doxPlayer(randomPlayer)
    end
end
